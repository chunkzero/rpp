## `build/types.rs`

```rust
use std::path::PathBuf;
use std::time::Duration;
use serde::{Deserialize, Serialize};

/// Fingerprint for cache invalidation.
/// Uses mtime + size for fast check, hash for content verification.
#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
pub struct Fingerprint {
    pub mtime: u64,
    pub size: u64,
    pub hash: u64, // xxhash3_64
}

/// A file entry discovered during the discovery phase.
#[derive(Debug, Clone)]
pub struct FileEntry {
    /// Absolute path to source file
    pub source_path: PathBuf,
    /// Path relative to source root (used for output)
    pub relative_path: PathBuf,
    /// Fingerprint for caching
    pub fingerprint: Fingerprint,
    /// File content (loaded lazily or eagerly based on cache status)
    pub content: Vec<u8>,
}

/// Index of files discovered during discovery phase.
#[derive(Debug, Default)]
pub struct FileIndex {
    /// Files that need processing (changed or new)
    pub entries: Vec<FileEntry>,
    /// Files that can be copied from cache (unchanged)
    pub cached: Vec<CachedEntry>,
}

/// Reference to a cached file that doesn't need reprocessing.
#[derive(Debug, Clone)]
pub struct CachedEntry {
    pub source_path: PathBuf,
    pub output_path: PathBuf,
    /// Cached processed content (stored to avoid reprocessing)
    pub content: Vec<u8>,
}

/// Record of a transformation applied by a processor.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Transformation {
    /// Processor name
    pub processor: String,
    /// Processor version at time of transformation
    pub version: String,
    /// xxhash3 of input content
    pub input_hash: u64,
    /// xxhash3 of output content
    pub output_hash: u64,
}

/// A file after processing through the processor chain.
#[derive(Debug, Clone)]
pub struct ProcessedFile {
    /// Original source path
    pub source_path: PathBuf,
    /// Output path (may differ if processor changed it)
    pub output_path: PathBuf,
    /// Processed content
    pub content: Vec<u8>,
    /// Chain of transformations applied
    pub transformations: Vec<Transformation>,
}

/// A file generated by a generator plugin.
#[derive(Debug, Clone)]
pub struct GeneratedFile {
    /// Output path relative to output root
    pub path: PathBuf,
    /// Generated content
    pub content: Vec<u8>,
    /// Generator that created this file
    pub generator: String,
}

/// Result of a complete build operation.
#[derive(Debug, Clone)]
pub struct BuildResult {
    /// Number of files processed through pipeline
    pub files_processed: usize,
    /// Number of files copied from cache
    pub files_cached: usize,
    /// Number of files created by generators
    pub files_generated: usize,
    /// Number of files cancelled by processors
    pub files_cancelled: usize,
    /// Total build duration
    pub duration: Duration,
}
```

## `build/error.rs`

```rust
use std::path::PathBuf;
use thiserror::Error;

#[derive(Debug, Error)]
pub enum BuildError {
    #[error("Discovery error: {0}")]
    Discovery(String),

    #[error("Failed to read file {path}: {source}")]
    FileRead {
        path: PathBuf,
        #[source]
        source: std::io::Error,
    },

    #[error("Failed to write file {path}: {source}")]
    FileWrite {
        path: PathBuf,
        #[source]
        source: std::io::Error,
    },

    #[error("Processor '{processor}' failed on {path}: {message}")]
    ProcessorFailed {
        processor: String,
        path: PathBuf,
        message: String,
    },

    #[error("Generator '{generator}' failed: {message}")]
    GeneratorFailed {
        generator: String,
        message: String,
    },

    #[error("Cache error: {0}")]
    Cache(String),

    #[error("Worker error: {0}")]
    Worker(String),

    #[error("Plugin error: {0}")]
    Plugin(String),
}
```

## `plugin/processor.rs`

```rust
use std::path::{Path, PathBuf};
use crate::build::BuildError;
use super::Plugin;

/// Context passed to processor plugins during file processing.
pub struct ProcessingContext<'a> {
    /// Relative path of the file being processed
    pub path: &'a Path,
    /// Current file content (may have been modified by previous processors)
    pub content: &'a [u8],
    /// Original source path
    pub source_path: &'a Path,
    /// Plugin configuration from rpp.toml
    pub config: &'a toml::Value,
}

/// Result returned by a processor after processing a file.
#[derive(Debug)]
pub enum ProcessResult {
    /// Continue processing with new content.
    /// Optionally change the output path.
    Continue {
        content: Vec<u8>,
        output_path: Option<PathBuf>,
    },

    /// Skip this processor, continue with next in chain.
    /// Content remains unchanged.
    Skip,

    /// Cancel processing this file entirely.
    /// File will not appear in output.
    Cancel,
}

impl ProcessResult {
    /// Create a Continue result with just new content.
    pub fn continue_with(content: Vec<u8>) -> Self {
        Self::Continue {
            content,
            output_path: None,
        }
    }

    /// Create a Continue result that also changes the output path.
    pub fn continue_with_path(content: Vec<u8>, path: PathBuf) -> Self {
        Self::Continue {
            content,
            output_path: Some(path),
        }
    }
}

/// Trait for processor plugins that transform file content.
pub trait ProcessorPlugin: Plugin {
    /// Glob patterns this processor handles (e.g., ["*.json", "**/*.mcmeta"]).
    fn patterns(&self) -> &[String];

    /// Priority for ordering in the processor chain.
    /// Lower values run first. Default is 100.
    fn priority(&self) -> i32 {
        100
    }

    /// Process a file and return the result.
    fn process(&self, ctx: &ProcessingContext) -> Result<ProcessResult, BuildError>;
}
```

```rust
use std::path::Path;
use crate::build::{BuildError, GeneratedFile, ProcessedFile};
use super::Plugin;

/// Context passed to generator plugins.
pub struct GeneratorContext<'a> {
    /// All files that have been processed (available for inspection)
    pub processed_files: &'a [ProcessedFile],
    /// Plugin configuration from rpp.toml
    pub config: &'a toml::Value,
    /// Output directory root
    pub output_dir: &'a Path,
}

/// Trait for generator plugins that create new files.
pub trait GeneratorPlugin: Plugin {
    /// Generate new files based on processed content.
    /// Called after all processors have run.
    fn generate(&self, ctx: &GeneratorContext) -> Result<Vec<GeneratedFile>, BuildError>;
}
```

## Tests

```rust
#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_fingerprint_equality() {
        let f1 = Fingerprint { mtime: 100, size: 200, hash: 300 };
        let f2 = Fingerprint { mtime: 100, size: 200, hash: 300 };
        let f3 = Fingerprint { mtime: 100, size: 200, hash: 301 };
        assert_eq!(f1, f2);
        assert_ne!(f1, f3);
    }

    #[test]
    fn test_process_result_constructors() {
        let content = vec![1, 2, 3];
        let result = ProcessResult::continue_with(content.clone());
        match result {
            ProcessResult::Continue { content: c, output_path } => {
                assert_eq!(c, content);
                assert!(output_path.is_none());
            }
            _ => panic!("Expected Continue"),
        }
    }
}
```
